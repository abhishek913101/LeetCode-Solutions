class Solution:
    def numSpecial(self, mat: List[List[int]]) -> int:
        m = len(mat)
        n = len(mat[0])
        ans = 0
        
        row_ones = [0] * m
        col_ones = [0] * n
        
        for i in range(m):
            for j in range(n):
                if mat[i][j] == 1:
                    row_ones[i] += 1
                    col_ones[j] += 1
        for i in range(m):
            for j in range(n):
                if mat[i][j] == 1 and row_ones[i] == 1 and col_ones[j] == 1:
                    ans += 1
                    
        return ans
