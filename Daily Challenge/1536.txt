class Solution:
    def minSwaps(self, grid: List[List[int]]) -> int:
        n = len(grid)
        # Store the number of trailing zeros for each row
        suffixZeros = [0] * n
        for i in range(n):
            count = 0
            for j in range(n - 1, -1, -1):
                if grid[i][j] == 0:
                    count += 1
                else:
                    break
            suffixZeros[i] = count
        
        ans = 0
        for i in range(n):
            neededZeros = n - 1 - i
            found_idx = -1
            for j in range(i, n):
                if suffixZeros[j] >= neededZeros:
                    found_idx = j
                    break
            
            if found_idx == -1:
                return -1
            for k in range(found_idx, i, -1):
                suffixZeros[k], suffixZeros[k-1] = suffixZeros[k-1], suffixZeros[k]
            
            ans += (found_idx - i)
            
        return ans
