class Solution:
    def generateString(self, str1: str, str2: str) -> str:
        n = len(str1)
        m = len(str2)
        result = ['?'] * (n + m - 1)
        fixed = [False] * (n + m - 1)
        for i in range(n):
            if str1[i] == 'T':
                for j in range(m):
                    if result[i + j] != '?' and result[i + j] != str2[j]:
                        return ""
                    result[i + j] = str2[j]
                    fixed[i + j] = True
        for i in range(len(result)):
            if result[i] == '?':
                result[i] = 'a'
        
        for i in range(n):
            if str1[i] == 'F':
                matches = True
                for j in range(m):
                    if result[i + j] != str2[j]:
                        matches = False
                        break
                if matches:
                    changed = False
                    for j in range(m - 1, -1, -1):
                        if not fixed[i + j]:
                            
                            if result[i + j] == 'a':
                                result[i + j] = 'b'
                                changed = True
                                break
                    if not changed:
                        return "" 

        return "".join(result)
