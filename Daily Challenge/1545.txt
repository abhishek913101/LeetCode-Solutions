class Solution:
    def findKthBit(self, n: int, k: int) -> str:
        if n == 1:
            return '0'
        midIndex = 1 << (n - 1)
        
        if k == midIndex:
            return '1'
        elif k < midIndex:
            return self.findKthBit(n - 1, k)
        else:
            mirror_k = 2 * midIndex - k
            return '1' if self.findKthBit(n - 1, mirror_k) == '0' else '0'
