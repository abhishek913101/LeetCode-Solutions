class Solution:
    def maxProductPath(self, grid: List[List[int]]) -> int:
        MOD = 1_000_000_007
        m = len(grid)
        n = len(grid[0])
        dp_max = [[0] * n for _ in range(m)]
        dp_min = [[0] * n for _ in range(m)]

        dp_max[0][0] = dp_min[0][0] = grid[0][0]

        for i in range(1, m):
            dp_max[i][0] = dp_min[i][0] = dp_max[i-1][0] * grid[i][0]
        for j in range(1, n):
            dp_max[0][j] = dp_min[0][j] = dp_max[0][j-1] * grid[0][j]

        for i in range(1, m):
            for j in range(1, n):
                val = grid[i][j]
                p1 = dp_max[i-1][j] * val
                p2 = dp_min[i-1][j] * val
                p3 = dp_max[i][j-1] * val
                p4 = dp_min[i][j-1] * val

                dp_max[i][j] = max(p1, p2, p3, p4)
                dp_min[i][j] = min(p1, p2, p3, p4)

        result = dp_max[m-1][n-1]
        return result % MOD if result >= 0 else -1
