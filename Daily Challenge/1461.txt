class Solution:
    def hasAllCodes(self, s: str, k: int) -> bool:
        required_codes_count = 1 << k
        found_codes = set()
        for i in range(len(s) - k + 1):
            substring = s[i:i+k]
            found_codes.add(substring)
            
            if len(found_codes) == required_codes_count:
                return True
        return len(found_codes) == required_codes_count
