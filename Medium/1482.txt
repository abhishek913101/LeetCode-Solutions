if m * k > len(bloomDay):
        return -1

    left = min(bloomDay)
    right = max(bloomDay)
    ans = -1

    while left <= right:
        mid = (left + right) // 2

        bouquets = 0
        flowers = 0

        for d in bloomDay:
            if d <= mid:
                flowers += 1
                if flowers == k:
                    bouquets += 1
                    flowers = 0
            else:
                flowers = 0

        if bouquets >= m:
            ans = mid
            right = mid - 1
        else:
            left = mid + 1

    return ans
